After loading user and checking ATIVO:

```javascript
async function authJwt(req, res, next) {
  const token = req.cookies?.access_token;
  if (!token) return res.status(401).json({ error: 'Não autenticado' });
  try {
    const payload = verifyAccess(token);
    const user = await prisma.user.findUnique({ where: { id: payload.sub } });
    if (!user || user.status !== 'ATIVO') return res.status(401).json({ error: 'Usuário inválido' });
    // SES-02: mustChangePassword gate — allowlist before 403
    if (user.mustChangePassword) {
      const allowlist = ['/api/auth/change-password', '/api/auth/me', '/api/auth/refresh', '/api/auth/logout'];
      const url = req.originalUrl || req.path || '';
      // originalUrl includes /api prefix when mounted under /api/auth, req.path is route-relative — check both
      const isAllowlisted = allowlist.some(p => url.startsWith(p) || req.path.startsWith(p.replace('/api/auth','')) || req.path.startsWith(p));
      // Simpler: check req.originalUrl against allowlist + also allow /api/auth/change-password with and without /api prefix
      // Robust: normalize to originalUrl and path
      if (!allowlist.some(p => url.startsWith(p))) {
        // Also allow direct path without /api prefix if app mounts differently in tests (createApp vs server)
        const pathOnly = req.path || '';
        const isPathAllowlisted = ['/change-password', '/me', '/refresh', '/logout'].some(s => pathOnly === s || pathOnly.startsWith(s + '/'));
        if (!isPathAllowlisted) {
          return res.status(403).json({ error: 'Troca de senha obrigatória' });
        }
      }
    }
    req.user = user;
    next();
  } catch { return res.status(401).json({ error: 'Sessão expirada' }); }
}
```

Simplify to robust check:
```javascript
if (user.mustChangePassword) {
  const allowed = ['/api/auth/change-password', '/api/auth/me', '/api/auth/refresh', '/api/auth/logout', '/change-password', '/me', '/refresh', '/logout'];
  const url = req.originalUrl || req.path;
  if (!allowed.some(p => url.includes(p))) {
    return res.status(403).json({ error: 'Troca de senha obrigatória' });
  }
}
```

But prefer exact prefix to avoid false allow (e.g., /api/reports/change-password shouldn't be allowlisted). Use:

```javascript
if (user.mustChangePassword) {
  const allowPaths = ['/api/auth/change-password', '/api/auth/me', '/api/auth/refresh', '/api/auth/logout'];
  const url = req.originalUrl || '';
  const isAllowlisted = allowPaths.some(p => url.startsWith(p));
  if (!isAllowlisted) {
    return res.status(403).json({ error: 'Troca de senha obrigatória' });
  }
}
```

This works for supertest where originalUrl is /api/auth/... and for direct app factory.

After, re-run SES-02 backend tests — non-allowlisted should now 403, allowlisted still 200, change clears flag.
