Modify backend/src/jobs/scheduler.js startScheduler() to call verify warn-only:

Inside startScheduler(), before creating intervals, add:

```javascript
if (env.smtpEnabled && env.smtp.host) {
  try {
    const transporter = require('../services/emailService').getTransporter?.();
    // getTransporter creates transporter lazily; verify returns promise
    if (transporter && typeof transporter.verify === 'function') {
      transporter.verify().catch(e => logger.warn({err: e.message}, 'SMTP verify failed — queue will retry'));
    } else {
      // Fallback: createTransport directly and verify (if getTransporter not exported, require nodemailer)
      const nodemailer = require('nodemailer');
      const t = nodemailer.createTransport({host: env.smtp.host, port: env.smtp.port, secure: env.smtp.secure, auth: env.smtp.user ? {user: env.smtp.user, pass: env.smtp.pass} : undefined});
      t.verify().catch(e => logger.warn({err: e.message}, 'SMTP verify failed — queue will retry'));
    }
  } catch(e){ logger.warn({err:e.message}, 'SMTP verify setup failed'); }
}
```

Note: emailService.js already has getTransporter() — ensure it is exported (it is: module.exports = {enqueue, processQueue} — need to add getTransporter to exports OR keep fallback). Prefer adding getTransporter to exports (small change).
